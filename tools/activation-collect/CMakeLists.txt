cmake_minimum_required(VERSION 3.20)
project(activation_collect LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_CPP_ROOT "" CACHE PATH "Required llama.cpp source tree")
set(LLAMA_CPP_LIB_DIR "" CACHE PATH "Required prebuilt llama.cpp library directory")
foreach(var LLAMA_CPP_ROOT LLAMA_CPP_LIB_DIR)
  if(NOT ${var} OR NOT IS_DIRECTORY "${${var}}")
    message(FATAL_ERROR "${var} must name an existing directory")
  endif()
endforeach()
execute_process(COMMAND git -C "${LLAMA_CPP_ROOT}" rev-parse HEAD OUTPUT_VARIABLE LLAMA_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
add_library(collect_core core.cpp "${LLAMA_CPP_ROOT}/vendor/hash/sha256/sha256.c")
target_include_directories(collect_core PUBLIC "${LLAMA_CPP_ROOT}/vendor" "${LLAMA_CPP_ROOT}/vendor/hash" "${LLAMA_CPP_ROOT}/ggml/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(collect_core PRIVATE -Wall -Wextra -Wpedantic)
foreach(lib llama llama-common ggml ggml-base)
  string(REPLACE "-" "_" var "${lib}")
  find_library(${var}_LIB NAMES ${lib} PATHS "${LLAMA_CPP_LIB_DIR}" NO_DEFAULT_PATH REQUIRED)
  list(APPEND LLAMA_LIBS "${${var}_LIB}")
endforeach()
add_executable(activation-collect main.cpp)
target_include_directories(activation-collect PRIVATE "${LLAMA_CPP_ROOT}/include" "${LLAMA_CPP_ROOT}/common")
target_compile_definitions(activation-collect PRIVATE LLAMA_HEAD="${LLAMA_HEAD}")
target_compile_options(activation-collect PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(activation-collect PRIVATE collect_core ${LLAMA_LIBS})
set_target_properties(activation-collect PROPERTIES BUILD_RPATH "${LLAMA_CPP_LIB_DIR}")
enable_testing()
add_executable(collect-tests tests.cpp)
target_link_libraries(collect-tests PRIVATE collect_core)
add_test(NAME collect-core COMMAND collect-tests)
